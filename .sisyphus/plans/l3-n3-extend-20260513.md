  ```python
  # 关键 imports
  from novel_analyzer.skills.assets import render_skill_prompt
  from novel_analyzer.llm.client import build_chat_model
  from novel_analyzer.config.settings import Settings
  from novel_analyzer.database.session import create_session_factory
  from sqlalchemy import text
  import json, re, time
  from pathlib import Path
  
  BRANCH = "e5becabd-e2f3-4045-9249-fa91f382dc9a"
  CHAPTERS = [10, 16, 44]
  
  # noise classifier (same as §7)
  NEGATION_VERB_PREFIXES = ('没','有','是','不','在','让','走','进','出','去','来','把','被','所','使','于','与','及','而','又','也','只','都','才','就')
  ORDINAL_RE = re.compile(r'^第[一二三四五六七八九十百千零〇\d]+(章|节|回|卷|话|篇|页|部分)?$')
  ONOMATOPOEIA_RE = re.compile(r'(.)\1{2,}')
  
  def classify(t):
      if not t or len(t) > 20: return 'malformed'
      if ORDINAL_RE.match(t): return 'ordinal'
      if ONOMATOPOEIA_RE.search(t): return 'onomatopoeia'
      if any(t.startswith(p) for p in NEGATION_VERB_PREFIXES) and len(t) <= 5: return 'verb_or_negation'
      if t.endswith(('与','和','及','了','着','过','的')) and len(t) <= 4: return 'truncated_tail'
      if len(t) == 1: return 'single_char'
      return 'likely_valid'
  
  # extract first JSON object from LLM response
  def extract_json(s: str):
      first = s.find('{')
      last = s.rfind('}')
      if first < 0 or last < 0: raise ValueError("no JSON found")
      return json.loads(s[first:last+1])
  
  # --- main ---
  factory = create_session_factory(Settings())
  with factory() as session:
      # count chapter_artifacts BEFORE
      before = session.execute(text("SELECT COUNT(*) FROM chapter_artifacts WHERE branch_id=:b"), {"b": BRANCH}).scalar()
      print(f"chapter_artifacts before: {before}")
      
      # load stored entities + cleaned_text for each chapter
      chapters_data = {}
      for ch in CHAPTERS:
          ca = session.execute(text("""
              SELECT payload_json FROM chapter_artifacts
              WHERE branch_id=:b AND chapter_index=:c AND artifact_type='chapter_analysis'
          """), {"b": BRANCH, "c": ch}).fetchone()
          ro = session.execute(text("""
              SELECT parsed_json FROM chapter_raw_outputs
              WHERE branch_id=:b AND chapter_index=:c AND parse_status='ok'
              ORDER BY created_at DESC LIMIT 1
          """), {"b": BRANCH, "c": ch}).fetchone()
          
          if not ca or not ro:
              chapters_data[ch] = {"error": f"missing data: ca={bool(ca)} ro={bool(ro)}"}
              continue
          
          stored_payload = ca[0] if isinstance(ca[0], dict) else json.loads(ca[0])
          parsed = ro[0] if isinstance(ro[0], dict) else json.loads(ro[0])
          
          stored_entities = stored_payload.get('key_entities') or []
          intake = parsed.get('intake') or {}
          cleaned_text = intake.get('cleaned_text') or ''
          title = intake.get('normalized_title') or stored_payload.get('normalized_title') or f"第{ch}章"
          
          if not cleaned_text:
              chapters_data[ch] = {"error": "no cleaned_text in raw_outputs", "stored_entities": stored_entities}
              continue
          
          chapters_data[ch] = {
              "stored_entities": stored_entities,
              "cleaned_text_chars": len(cleaned_text),
              "cleaned_text": cleaned_text,
              "title": title,
          }
  
  # call LLM for each chapter that has cleaned_text
  model = build_chat_model()
  results = {"branch": BRANCH, "chapters": {}}
  
  for ch, data in chapters_data.items():
      if "error" in data:
          results["chapters"][ch] = data
          print(f"ch{ch}: SKIP ({data['error']})")
          continue
      
      payload = {
          "chapter_index": ch,
          "normalized_title": data["title"],
          "previous_summary": "",
          "prior_context_json": "{}",
          "graph_context_json": "{}",
          "state_summary_json": "{}",
          "chapter_content": data["cleaned_text"],
      }
      try:
          prompt = render_skill_prompt('chapter-intake-and-facts', payload)
          t0 = time.perf_counter()
          resp = model.invoke(prompt)
          elapsed = time.perf_counter() - t0
          raw = resp.content if hasattr(resp, 'content') else str(resp)
          parsed = extract_json(raw)
          chars = parsed.get('facts', {}).get('characters', [])
          new_entities = [c.get('label', '') for c in chars if c.get('label')]
          
          stored_classes = {}
          for e in data["stored_entities"]:
              stored_classes[classify(str(e))] = stored_classes.get(classify(str(e)), 0) + 1
          new_classes = {}
          for e in new_entities:
              new_classes[classify(str(e))] = new_classes.get(classify(str(e)), 0) + 1
          
          results["chapters"][ch] = {
              "stored_entities": data["stored_entities"],
              "new_entities": new_entities,
              "stored_classification": stored_classes,
              "new_classification": new_classes,
              "cleaned_text_chars": data["cleaned_text_chars"],
              "elapsed_sec": round(elapsed, 2),
              "noise_kept": len(new_entities) > 0 and any(c != 'likely_valid' for c in new_classes.keys()),
          }
          print(f"ch{ch}: stored={data['stored_entities']}")
          print(f"      new   ={new_entities}")
          print(f"      classes: {new_classes}")
      except Exception as e:
          results["chapters"][ch] = {"error": str(e), "stored_entities": data["stored_entities"]}
          print(f"ch{ch}: LLM FAIL: {e}")
  
  # verdict
  successful = [c for c in results["chapters"].values() if "error" not in c]
  noise_in_new = sum(1 for c in successful if c.get("noise_kept"))
  if not successful:
      verdict = "inconclusive (no chapters processed)"
  elif noise_in_new == 0:
      verdict = "hypothesis C primary (all new outputs clean)"
  elif noise_in_new == len(successful):
      verdict = "hypothesis A primary (all new outputs still noisy)"
  else:
      verdict = f"both contribute ({noise_in_new}/{len(successful)} new outputs still noisy)"
  results["verdict"] = verdict
  results["model"] = model.model_name if hasattr(model, 'model_name') else "unknown"
  print(f"\nVERDICT: {verdict}")
  
  # save
  out = Path(".sisyphus/evidence/l3-n3-dryrun-e5becabd-20260513.json")
  out.write_text(json.dumps(results, ensure_ascii=False, indent=2))
  print(f"Evidence: {out}")
  
  # verify no DB writes
  with factory() as session:
      after = session.execute(text("SELECT COUNT(*) FROM chapter_artifacts WHERE branch_id=:b"), {"b": BRANCH}).scalar()
      assert after == before, f"DB MUTATED: {before} → {after}"
      print(f"chapter_artifacts after: {after} (unchanged ✓)")
  ```
